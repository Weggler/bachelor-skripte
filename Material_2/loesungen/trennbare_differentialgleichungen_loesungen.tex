\begin{exercisebox}[Substitution] %Papula IV, Abschnitt 2, Aufgabe 3
	Lösen Sie das folgende Anfangswertproblem:
	$
	y' = \displaystyle{\frac{y}{x} } +\displaystyle{\frac{x}{y} }, \quad y(1) = \sqrt{2}\,.
	$

\vspace{0.3cm}
\textbf{Lösung:} \newline
Diese Aufgabe entspricht Aufg. \(3\) in dem Lehrbuch Papula Band 2, Abschnitt IV.2 zu finden auf Seite \(520\) und die Lösung finden Sie auf Seite \(726\).

\vspace{0.3cm}
Die Differentialgleichung hat die Form \[ y' = f\left( \frac{y}{x}\right)\,.\] Die Idee ist die Substitution $u=y/x$ durchzuführen und eine Differentialgleichung mit trennbaren Variablen bzgl. \(u\) herzuleiten:
\begin{enumerate}
	\item Die Substitution führt auf den folgenden Ausdruck für \(y'\)
\begin{eqnarray*}
	u=\frac{y}{x} \Rightarrow y = u\cdot x \Rightarrow y' = u'\cdot x + u
\end{eqnarray*}
\item Damit lautet die Differentialgleichung bzgl. \(u\)
	\begin{eqnarray*}
		u' \cdot x + u = u + \frac1u \Rightarrow u\cdot u' = \frac1x
	\end{eqnarray*}
\item Dies ist eine Differentialgleichung bzgl. \(u\), die sich durch Trennung der Variablen lösen läßt:
	\[
\begin{aligned}
\int u\,\mathrm du &= \int\frac1x\,\mathrm dx
&&\Rightarrow\quad \frac12u^2(x)=\ln|x|+c,\\
u(x)&=\pm\sqrt{2\ln|x|+2c},\\
y(x)&=x\,u(x)=\pm x\sqrt{2\ln|x|+2c}.
\end{aligned}
\]
\item Die Integrationskonstante \(c\) ist durch die Bedingung $y(1) = \sqrt{2}$ bestimmbar:
	\[y(1) = 1\cdot \sqrt{2\ln|1| + 2c} = \sqrt{2c} \stackrel{!}{=} \sqrt{2} \Rightarrow c=1\,.\,. \]
\item Die eindeutige Lösung des Anfangswertproblem lautet:  \[ y(x) = x\sqrt{2 (\ln|x| + 1) }= x\sqrt{2 (\ln|x| + \ln(e)) }= x\sqrt{2\ln|ex|} \]
	\end{enumerate}
\end{exercisebox}

\begin{exercisebox}[Logistisches Wachstum]
	Die Anzahl der von einem Virus infizierten Personen $x(t)$ zum Zeitpunkt $t$ in einer Bevölkerung der Größe $N$ kann folgendermaßen modelliert werden: Pro Tag habe jede Person $k>0$ zufällige Kontakte, wobei der Virus bei jedem Kontakt übertragen wird. Stellen Sie eine Differentialgleichung auf und ermitteln Sie die Lösung. Wie lautet die Lösung des zugehörigen Anfangswertproblemes wenn $x(0)=1$ war?

Gehen Sie Schritt für Schritt vor:
	\begin{itemize}
		\item[(a)] Wie viele nicht infizierte Personen gibt es in Abhängigkeit von $x(t)$ zum Zeitpunkt $t$? Wie groß ist der Anteil der nicht infizierten Personen zum Zeitpunkt $t$?
		\item[(b)] Wie viele Personen trifft eine infizierte Person in einer kleinen Zeitpanne $[t, t+\Delta t]$? Wieviele nicht infizierte Personen trifft eine infizierte Person in $[t, t+\Delta t]$?
		\item[(c)] Wie viele Kontakte von infizierten Personen mit nicht infizierten Personen gibt es insgesamt in $[t,t+\Delta t]$? Wie viele Personen werden daher in $[t, t+\Delta t]$ neu angesteckt?
		\item[(d)] Drücken Sie $x(t+\Delta t)$ durch $x(t)$ und die Anzahl der neu infizierten Personen in $[t, t+\Delta t]$ aus.
	\item[(e)] Führen Sie den Grenzwert $\Delta t\to 0$ aus, um eine Differentialgleichung zu erhalten.
\end{itemize}
\vspace{0.3cm}
\textbf{Lösung:} \newline
\begin{itemize}
	\item[(a)] $N-x(t), \; (N-x(t))/N $
	\item[(b)] $k\cdot \Delta t, \; k\cdot (N-x(t))/N \cdot \Delta t $
	\item[(c)] $x(t) \cdot k \cdot (N-x(t))/N\cdot \Delta t $
	\item[(d)] $x(t+\Delta t) = x(t) + x(t) \cdot k \cdot (N-x(t))/N\cdot \Delta t $
	\item[(e)] \[ \frac{x(t+\Delta t) - x(t)}{\Delta t} = x(t)\cdot  k \cdot (N-x(t))/N  \stackrel{\Delta t\to 0}{\to} \dot x = k\cdot x(N-x)/N \]
		Lösung durch Trennung der Variablen
		\[ \int \frac{1}{x(N-x)} \, \mathrm{d}x = \int \frac{k}{N}\,\mathrm{d}t \]
		Integration durch Partialbruchzerlegung
		\[ \int\frac{1}{x(N-x)} \, \mathrm{d}x = \int\left( \frac{1}{N\cdot x} - \frac{1}{N\cdot (N-x)} \right) \,\mathrm{d}x = \frac1N \left( \log(x) - \log(N-x)\right) = \frac1N \log\left( \frac{x}{N-x}\right) \]
		Also
		\[ \log\left( \frac{x}{N-x}\right) = k\cdot t + c \Leftrightarrow \frac{x}{N-x} = c\cdot e^{kt} \Leftrightarrow x(t) = N\cdot \frac{c\cdot e^{kt}}{1+c\cdot e^{kt}} \]
		Lösung des Anfangswertproblems $x$ mit $c= \frac{1}{n-1}$.
\end{itemize}
\end{exercisebox}
